def generate_copy(
    db,
    user_id: int,
    portfolio: dict,
    format_type: str = "sahibinden"
) -> Tuple[bool, str]:
    """
    İlan metni üret.
    Returns: (success: bool, text: str)
    """
    cost = db.get_credit_cost("ai_copywriter")
    if not db.deduct_credits(user_id, cost):
        return False, f"Yetersiz kredi! Bu işlem {cost} kredi gerektirir."

    api_key = db.get_gemini_api_key() or ""
    if not api_key:
        db.add_credits(user_id, cost)  # Krediyi iade et
        return False, "Gemini API anahtarı system_management tablosunda ayarlanmamış."

    try:
        import google.generativeai as genai
        from google.api_core import exceptions
        import time
        genai.configure(api_key=api_key)
        
        # Model adını DB'den al
        model_name = db.get_gemini_model()
        model = genai.GenerativeModel(model_name)
        
        format_tercihi = FORMAT_MAPPING.get(format_type, "Format/Stil Tercihi: Standart")
        ozellikler = _portfolio_to_prompt(portfolio)
        prompt = OPTIMIZED_PROMPT_TEMPLATE.format(
            ozellikler=ozellikler,
            format_tercihi=format_tercihi
        ).strip()

        # Retry Logic & Throttling
        max_retries = 3
        for attempt in range(max_retries):
            try:
                time.sleep(0.5) # Throttling
                response = model.generate_content(prompt)
                text = response.text.strip()
                log.info(f"AI Copywriter başarılı — Format: {format_type}, Kullanıcı: {user_id}")
                return True, text
            except exceptions.ResourceExhausted:
                if attempt < max_retries - 1:
                    log.warning(f"⚠️ Gemini Rate Limit doldu, 10 saniye bekleniyor... (Deneme {attempt+1})")
                    time.sleep(10)
                    continue
                else: raise
            except Exception as e:
                log.error(f"Gemini API hatası: {e}")
                break

    except Exception as e:
        db.add_credits(user_id, cost)  # Hata durumunda iade
        err_msg = str(e).lower()
        if "429" in err_msg or "quota" in err_msg:
            return False, "Çok fazla istek atıldı (Rate Limit) veya günlük kota doldu. Lütfen 1 dakika bekleyip tekrar deneyin. Krediniz iade edildi."

            return False, "Geçersiz API anahtarı. Krediniz iade edildi. Lütfen ayarlarınızı kontrol edin."
        
        log.error(f"AI Copywriter hatası: {e}")
        return False, f"API Hatası: {str(e)}"

fsbo_scraper.py — FSBO (Sahibinden Satılık) İlan Avcısı

Strateji:

QThread tabanlı çalışır — UI thread'i bloklamaz.

import logging
import time
import re
from datetime import datetime
from PyQt6.QtCore import QObject, pyqtSignal

log = logging.getLogger(__name__)

PROPERTY_TYPES = ["daire", "villa", "arsa", "dükkan", "ofis", "müstakil ev"]
SOURCE_SITES = ["hepsiemlak.com", "zingat.com", "emlakjet.com"]


# ── Yardımcı: URL oluştur ────────────────────────────────────────────────────

def _build_google_url(district: str, prop_type: str, listing_type: str = "satılık") -> str:
    """Google arama URL'si üret."""
    query = f'sahibinden {listing_type} {prop_type} {district} site:hepsiemlak.com OR site:zingat.com'
    from urllib.parse import quote_plus
    return f"https://www.google.com/search?q={quote_plus(query)}&num=20&hl=tr"


def _build_hepsiemlak_url(district: str, prop_type: str) -> str:
    """Hepsiemlak için doğrudan URL (sahibinden ilanları için filtre)."""
    from urllib.parse import quote_plus
    # Hepsiemlak oda sahibi ilanları
    q = f"{prop_type} {district}".strip().lower()
    return f"https://www.hepsiemlak.com/satilik?query={quote_plus(q)}"


# ── Ana Scraper Fonksiyonu ────────────────────────────────────────────────────

def scrape_fsbo(district: str, prop_type: str, listing_type: str = "satılık",
                max_results: int = 15) -> list:
    """
    FSBO ilanlarını tara ve liste döndür.
    Her eleman: {title, source_url, location, price_text, raw_snippet, source_site}
    """
    results = []

    try:
        import requests
        from bs4 import BeautifulSoup

        headers = {
            "User-Agent": (
                "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                "AppleWebKit/537.36 (KHTML, like Gecko) "
                "Chrome/124.0.0.0 Safari/537.36"
            ),
            "Accept-Language": "tr-TR,tr;q=0.9",
            "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
        }

        # ── 1. Hepsiemlak Denemesi ─────────────────────────────────────────
        try:
            url = _build_hepsiemlak_url(district, prop_type)
            resp = requests.get(url, headers=headers, timeout=12)
            resp.raise_for_status()
            soup = BeautifulSoup(resp.text, "html.parser")

            cards = soup.select(".list-view-content, .card-ilan, [class*='listing']")
            for card in cards[:max_results]:
                title_el = card.select_one("h3, h2, [class*='title']")
                price_el  = card.select_one("[class*='price'], [class*='fiyat']")
                loc_el    = card.select_one("[class*='location'], [class*='konum']")
                link_el   = card.select_one("a[href]")

                title = title_el.get_text(strip=True) if title_el else f"{district} {prop_type}"
                price = price_el.get_text(strip=True) if price_el else "—"
                loc   = loc_el.get_text(strip=True) if loc_el else district
                href  = link_el["href"] if link_el else url
                if href and not href.startswith("http"):
                    href = "https://www.hepsiemlak.com" + href

                snippet = card.get_text(" ", strip=True)[:200]

                if title and len(title) > 3:
                    results.append({
                        "title": title,
                        "source_url": href,
                        "location": loc or district,
                        "price_text": price,
                        "raw_snippet": snippet,
                        "source_site": "hepsiemlak",
                    })

            log.info(f"Hepsiemlak: {len(results)} ilan bulundu.")

        except Exception as e:
            log.warning(f"Hepsiemlak tarama hatası: {e}")

        time.sleep(1.5)  # Rate-limit önlemi

        # ── 2. Google Snippet Denemesi ─────────────────────────────────────
        if len(results) < 5:
            try:
                gurl = _build_google_url(district, prop_type, listing_type)
                resp2 = requests.get(gurl, headers=headers, timeout=12)
                soup2 = BeautifulSoup(resp2.text, "html.parser")

                for g in soup2.select("div.g")[:max_results]:
                    a = g.select_one("a[href]")
                    h3 = g.select_one("h3")
                    desc = g.select_one(".VwiC3b, .s3v9rd, [class*='snippet']")

                    if not a or not h3:
                        continue
                    href = a.get("href", "")
                    if href.startswith("/url?q="):
                        href = href[7:].split("&")[0]
                    title = h3.get_text(strip=True)
                    snippet = desc.get_text(strip=True) if desc else ""

                    # Fiyat regex
                    price_match = re.search(r"[\d\.]+\s*(?:TL|₺|bin)", snippet, re.IGNORECASE)
                    price_text = price_match.group() if price_match else "—"

                    site = next((s for s in SOURCE_SITES if s in href), "google")

                    results.append({
                        "title": title,
                        "source_url": href,
                        "location": district,
                        "price_text": price_text,
                        "raw_snippet": snippet[:200],
                        "source_site": site,
                    })

                log.info(f"Google: {len(results)} toplam ilan.")

            except Exception as e:
                log.warning(f"Google tarama hatası: {e}")

    except ImportError:
        log.warning("requests veya beautifulsoup4 kurulu değil — demo mod aktif.")

    # ── Demo/Fallback: Hiç sonuç bulunamazsa örnek veri ──────────────────────
    if not results:
        results = _demo_results(district, prop_type)

    return results[:max_results]


def _demo_results(district: str, prop_type: str) -> list:
    """
    Tarama başarısız olursa kullanıcıya nasıl göründüğünü gösteren
    örnek (demo) kayıtlar döndür.
    """
    base_prices = [1_850_000, 2_200_000, 975_000, 3_100_000, 1_450_000]

    return [

class FSBOWorker(QObject):
    """Tarama işlemini arka planda yapar."""
    finished = pyqtSignal(list)   # results list
    error    = pyqtSignal(str)
    progress = pyqtSignal(str)    # durum mesajı

    def __init__(self, district: str, prop_type: str,
                 listing_type: str = "satılık", max_results: int = 15):
        super().__init__()
        self.district = district
        self.prop_type = prop_type
        self.listing_type = listing_type
        self.max_results = max_results

    def run(self):
        try:
            self.progress.emit(f"🔍 {self.district} bölgesinde '{self.prop_type}' aranıyor...")
            results = scrape_fsbo(
                self.district, self.prop_type,
                self.listing_type, self.max_results
            )
            self.progress.emit(f"✅ {len(results)} ilan bulundu.")
            self.finished.emit(results)
        except Exception as e:
            self.error.emit(str(e))
